<template>
  <Head><title>{{ $t('leaves.heading') }} - MaterialTrack</title></Head>

  <main class="p-6">
    <div class="flex justify-between items-center mb-6">
      <h2 class="text-2xl font-bold text-gray-900 flex items-center gap-2">
        <i class="fa fa-calendar-minus text-indigo-600"></i>
        {{ $t('leaves.heading') }}
      </h2>
      <Link
        v-if="hasAnyPermission(['leaves.create'])"
        href="/apps/leaves/create"
        class="btn-modern btn-modern-primary"
      >
        <i class="fa fa-plus me-2"></i> {{ $t('common.new') }}
      </Link>
    </div>

    <DataTable
      :columns="columns"
      :rows="leaves.data"
      :pagination="leaves"
      row-key="id"
      :striped="true"
      :loading="false"
      :empty-text="$t('leaves.empty')"
    >
      <template #cell-employee="{ row }">
        {{ row.employee?.first_name }} {{ row.employee?.last_name || '' }}
      </template>
      <template #cell-leave_type="{ row }">
        {{ row.leave_type?.name }}
      </template>
      <template #cell-date_range="{ row }">
        {{ row.start_date }} s/d {{ row.end_date }}
      </template>
      <template #cell-total_days="{ row }">
        <code>{{ row.total_days }} {{ $t('leaves.col_days') }}</code>
      </template>
      <template #cell-status="{ row }">
        <span
          class="px-2 py-1 text-xs rounded-full font-bold"
          :class="{
            'bg-yellow-100 text-yellow-800': row.status === 'pending',
            'bg-green-100 text-green-800': row.status === 'approved',
            'bg-red-100 text-red-800': row.status === 'rejected'
          }"
        >
          {{ row.status }}
        </span>
      </template>
      <template #cell-actions="{ row }">
        <div class="flex items-center justify-center gap-2" v-if="row.status === 'pending'">
          <button v-if="hasAnyPermission(['leaves.approve'])" class="act-btn act-btn-success" @click="approve(row.id)">
            <i class="fa fa-check"></i>
          </button>
          <button v-if="hasAnyPermission(['leaves.reject'])" class="act-btn act-btn-danger" @click="reject(row.id)">
            <i class="fa fa-times"></i>
          </button>
        </div>
        <span v-else class="text-muted small">-</span>
      </template>
    </DataTable>
  </main>
</template>

<script>
import LayoutApp from "@/Layouts/App.vue";
import DataTable from "@/Components/DataTable.vue";
import Pagination from "@/Components/Pagination.vue";
import { Head, Link, router } from "@inertiajs/vue3";
import { ref } from "vue";
import { useI18n } from "vue-i18n";

export default {
  layout: LayoutApp,
  components: { Head, Link, DataTable, Pagination },
  props: { leaves: Object },
  setup() {
    const { t } = useI18n();
    const columns = ref([
      { key: "employee", label: t("leaves.col_employee") },
      { key: "leave_type", label: t("leaves.col_type") },
      { key: "date_range", label: t("leaves.col_date") },
      { key: "total_days", label: t("leaves.col_days"), align: "center" },
      { key: "status", label: t("leaves.col_status"), align: "center" },
      { key: "actions", label: t("common.action"), align: "center", width: "150px" },
    ]);
    const approve = (id) => {
      if (confirm(t("leaves.approve_confirm") || "Setujui pengajuan cuti ini?")) {
        router.post(`/apps/leaves/${id}/approve`);
      }
    };
    const reject = (id) => {
      const reason = prompt("Alasan penolakan:");
      if (reason) {
        router.post(`/apps/leaves/${id}/reject`, { rejection_reason: reason });
      }
    };
    return { approve, reject, columns };
  },
};
</script>
