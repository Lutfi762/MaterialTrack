<template>
  <Head><title>{{ $t('attendances.heading') }} - MaterialTrack</title></Head>

  <main class="p-6">
    <div class="flex justify-between items-center mb-6">
      <h2 class="text-2xl font-bold text-gray-900 flex items-center gap-2">
        <i class="fa fa-user-check text-indigo-600"></i>
        {{ $t('attendances.heading') }}
      </h2>
      <Link
        v-if="hasAnyPermission(['attendances.create'])"
        href="/apps/attendances/create"
        class="btn-modern btn-modern-primary"
      >
        <i class="fa fa-plus me-2"></i> {{ $t('common.new') }}
      </Link>
    </div>

    <DataTable
      :columns="columns"
      :rows="attendances.data"
      :pagination="attendances"
      row-key="id"
      :striped="true"
      :loading="false"
      :empty-text="$t('attendances.empty')"
    >
      <template #cell-date="{ row }">
        {{ row.date }}
      </template>
      <template #cell-employee="{ row }">
        {{ row.employee?.first_name }} {{ row.employee?.last_name }}
      </template>
      <template #cell-clock_in="{ row }">
        {{ row.clock_in || '-' }}
      </template>
      <template #cell-clock_out="{ row }">
        {{ row.clock_out || '-' }}
      </template>
      <template #cell-status="{ row }">
        <span
          class="px-2 py-1 text-xs rounded-full font-bold"
          :class="{
            'bg-green-100 text-green-800': row.status === 'present',
            'bg-yellow-100 text-yellow-800': row.status === 'late',
            'bg-red-100 text-red-800': row.status === 'absent',
            'bg-blue-100 text-blue-800': row.status === 'leave'
          }"
        >
          {{ row.status }}
        </span>
      </template>
    </DataTable>
  </main>
</template>

<script>
import LayoutApp from "@/Layouts/App.vue";
import DataTable from "@/Components/DataTable.vue";
import Pagination from "@/Components/Pagination.vue";
import { Head, Link } from "@inertiajs/vue3";
import { ref } from "vue";
import { useI18n } from "vue-i18n";

export default {
  layout: LayoutApp,
  components: { Head, Link, DataTable, Pagination },
  props: { attendances: Object },
  setup() {
    const { t } = useI18n();
    const columns = ref([
      { key: "date", label: t("attendances.col_date") },
      { key: "employee", label: t("attendances.col_employee"), sortable: true },
      { key: "clock_in", label: t("attendances.col_check_in") },
      { key: "clock_out", label: t("attendances.col_check_out") },
      { key: "status", label: t("attendances.col_status"), align: "center" },
    ]);
    return { columns };
  },
};
</script>
